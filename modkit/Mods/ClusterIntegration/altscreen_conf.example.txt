# Copy to altscreen_conf.txt (next to Update/) to tune the CarPlay map in the
# cockpit (needs enable_carplay_altscreen.txt). One key=value per line, numbers
# only; '#' starts a comment. install.sh copies the valid lines to
# /mnt/app/eso/bin/apps/cluster/altscreen.conf; the hook reads that file every
# time the iPhone connects, so after the card ran, reconnecting is enough.
# An altscreen_conf.txt without any setting removes the installed file again.
#
# View areas: the part of the 1920x720 picture the cockpit shows, as
# x,y,width,height in pixels from the top left. The iPhone keeps its turn
# card, ETA line and speed limit sign inside that rectangle. "large" is the
# large-map layout, "small" the large-gauges layout. These are the defaults
# (Audi A6 C8, standard views):
#large=500,155,940,410
#small=624,158,672,400
#
# Cockpits with the dial in the centre ("Sport" layout, the "Performance"
# view of S and RS models, e-tron): the mod recognises that layout and uses
# these two areas instead of large and small. There the status bar is
# higher, and the small view is a window on the left. Defaults:
#sport_large=500,155,940,380
#sport_small=54,155,480,370
#
# A card or the arrival line is hidden at the bottom: lower the last number
# in steps of 20. Cut off on the left: raise the first number. Cut off on
# the right: lower the third number.
#
# Physical size of the picture in mm, as told to the iPhone. Larger numbers
# make the iPhone draw its cards and texts smaller.
#phys=384,144
#
# Steering-wheel roller: 1 = positive steps zoom out (A6 C8), 0 = the other
# way round.
#zoom_invert=1
#
# Remember the zoom across restarts (1/0) and the largest step count kept.
#zoom_restore=1
#zoom_max=10
#
# One zoom per cockpit view, large map and big gauges (1), or one for both (0).
#zoom_per_view=1
#
# Milliseconds between two zoom steps when the map moves to the other view's
# zoom after a view change (50 to 2000).
#zoom_step_ms=600
#
# Ask the iPhone for its speed limit sign on the cockpit map (1/0).
#speedlimit=1
#
# What the iPhone draws on the cockpit map besides the map itself: its arrival
# line (eta) and its turn card (turncard), 1/0 each. With Audi's own turn box
# in the cockpit (nav_box.txt) the same information is there twice; turncard=0
# and eta=0 leave the map alone.
#eta=1
#turncard=1
#
# Ports the iPhone may send the cockpit video to, tried in this order. Audi's
# firewall passes these three anyway. When none of them is free the mod uses
# 6031, which it has to open itself at every start; ports=0 goes straight to
# 6031 (as before beta 5).
#ports=6100,6200,7100
#
# A full picture (keyframe) is requested this often, in milliseconds
# (1000..60000). A damaged picture heals within that time.
#keyframe_ms=3000
